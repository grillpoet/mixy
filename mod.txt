from pathlib import Path

ROOT = Path(r"D:\pro\jvm\AndroidStudio\mixy")
BASE = ROOT / "app/src/main/java/com/grillpoet/mixy"

def patch(path: Path, old: str, new: str):
    text = path.read_text(encoding="utf-8")
    if old not in text:
        raise SystemExit(f"❌ 未找到目标：{path}\n---\n{old[:100]}\n---")
    path.write_text(text.replace(old, new, 1), encoding="utf-8")
    print(f"✅ {path}")
