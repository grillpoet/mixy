#!/usr/bin/env python3
"""MediaStore 查询过滤 IS_TRASHED / IS_PENDING"""
from pathlib import Path

ROOT = Path(r"D:\pro\jvm\AndroidStudio\mixy")
p = ROOT / "app/src/main/java/com/grillpoet/mixy/data/MediaStoreDataSource.kt"

def main():
    text = p.read_text(encoding="utf-8")

    # 加 import
    if "import android.os.Build" not in text:
        text = text.replace(
            "import android.net.Uri\n",
            "import android.net.Uri\nimport android.os.Build\n",
            1,
        )
    if "import android.provider.MediaStore" in text and "import androidx.annotation.RequiresApi" not in text:
        pass  # 不加 RequiresApi，用 runtime 判断即可

    # 在 queryFolders 里加过滤辅助
    anchor = '''    private fun queryAudio(): List<MediaItem> {
        val uri = MediaStore.Audio.Media.EXTERNAL_CONTENT_URI
        val proj = arrayOf(
            MediaStore.Audio.Media._ID,
            MediaStore.Audio.Media.DISPLAY_NAME,
            MediaStore.Audio.Media.MIME_TYPE,
            MediaStore.Audio.Media.SIZE,
            MediaStore.Audio.Media.DATE_MODIFIED,
            MediaStore.Audio.Media.DURATION,
            MediaStore.Audio.Media.BUCKET_DISPLAY_NAME,
            MediaStore.Audio.Media.DATA
        )
        val sel = "${MediaStore.Audio.Media.IS_MUSIC} != 0"'''
    new = '''    /**
     * 基础过滤：排除回收站项（Android 11+）和正在写入项。
     * 已删除项出现在列表里点开会失败，正在写入的缩略图加载失败。
     */
    private fun baseSelection(): String {
        val parts = mutableListOf<String>()
        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.R) {
            parts += "${MediaStore.MediaColumns.IS_TRASHED} = 0"
        }
        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.Q) {
            parts += "${MediaStore.MediaColumns.IS_PENDING} = 0"
        }
        return parts.joinToString(" AND ")
    }

    /** 拼接用户已有 selection 与基础过滤。 */
    private fun combineSelection(user: String?): String? {
        val base = baseSelection()
        return when {
            user.isNullOrBlank() -> base.ifBlank { null }
            base.isBlank() -> user
            else -> "($user) AND ($base)"
        }
    }

    private fun queryAudio(): List<MediaItem> {
        val uri = MediaStore.Audio.Media.EXTERNAL_CONTENT_URI
        val proj = arrayOf(
            MediaStore.Audio.Media._ID,
            MediaStore.Audio.Media.DISPLAY_NAME,
            MediaStore.Audio.Media.MIME_TYPE,
            MediaStore.Audio.Media.SIZE,
            MediaStore.Audio.Media.DATE_MODIFIED,
            MediaStore.Audio.Media.DURATION,
            MediaStore.Audio.Media.BUCKET_DISPLAY_NAME,
            MediaStore.Audio.Media.DATA
        )
        val sel = "${MediaStore.Audio.Media.IS_MUSIC} != 0"'''
    if anchor not in text:
        raise SystemExit("❌ 未找到 queryAudio 锚点")
    text = text.replace(anchor, new, 1)

    # 所有 query 调用点：把 sel 包一层 combineSelection
    # 4 处调用：query(uri, proj, sel) / query(uri, proj, null) × 3
    # 用泛化替换更稳：
    text = text.replace(
        "        return query(uri, proj, sel) { c ->",
        "        return query(uri, proj, combineSelection(sel)) { c ->",
        1,
    )
    # 其余 3 处 query(uri, proj, null)：不能全替换，需要只替换 queryImages / queryVideos
    # 由于这 3 处写法相同，用上下文定位
    text = text.replace(
        '''            MediaStore.Images.Media.DATA
        )
        return query(uri, proj, null) { c ->''',
        '''            MediaStore.Images.Media.DATA
        )
        return query(uri, proj, combineSelection(null)) { c ->''',
        1,
    )
    text = text.replace(
        '''            MediaStore.Video.Media.DATA
        )
        return query(uri, proj, null) { c ->''',
        '''            MediaStore.Video.Media.DATA
        )
        return query(uri, proj, combineSelection(null)) { c ->''',
        1,
    )

    p.write_text(text, encoding="utf-8")
    print(f"✅ {p}")

if __name__ == "__main__":
    main()
